# Completion event 1
communism.1 = {
	type = country_event
	placement = root
	title = communism.1.t
	desc = communism.1.desc
	flavor = communism.1.f

	duration = 3

	event_image = {
		video = "europenorthamerica_political_extremism"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_election.dds"

	trigger = {
		# triggered by communsim JE 1
	}

	immediate = {
		if = {
			limit = {
				any_interest_group = {
					law_stance = {
						law = law_type:law_council_republic
						value > neutral
					}
				}
			}
			random_interest_group = {
				limit = {
					law_stance = {
						law = law_type:law_council_republic
						value > neutral
					}
				}
				set_variable = communist_ig_var
				save_scope_as = unions
				leader = {
					save_scope_as = union_leader
				}
			}
		}
		else = {
			ig:ig_trade_unions ?= {
				set_variable = communist_ig_var
				save_scope_as = unions
				leader = {
					save_scope_as = union_leader
				}
			}
		}
	}

	option = { # wage controls
		name = communism.1.a
		trigger = {
			NOT = { has_law_or_variant = law_type:law_wage_subsidies }
			NOT = { has_law_or_variant = law_type:law_old_age_pension }
		}
		activate_law = law_type:law_wage_subsidies
		if = {
			limit = {
				any_interest_group = {
					has_variable = communist_ig_var
				}
			}
			custom_tooltip = communist_unions.tt
		}
	}
	option = { # ban slavery
		name = communism.1.b
		trigger = {
			NOT = { has_law = law_type:law_slavery_banned }
		}
		activate_law = law_type:law_slavery_banned
		if = {
			limit = {
				any_interest_group = {
					has_variable = communist_ig_var
				}
			}
			custom_tooltip = communist_unions.tt
		}
	}
	option = { # Regulatory bodies
		name = communism.1.c
		trigger = {
			NOT = { has_law_or_variant = law_type:law_regulatory_bodies }
			NOT = { has_law_or_variant = law_type:law_worker_protections }
		}
		activate_law = law_type:law_regulatory_bodies
		if = {
			limit = {
				any_interest_group = {
					has_variable = communist_ig_var
				}
			}
			custom_tooltip = communist_unions.tt
		}
	}
	option = {
		name = communism.1.d
		default_option = yes
		if = {
			limit = {
				any_interest_group = {
					has_variable = communist_ig_var
				}
			}
			custom_tooltip = communist_unions.tt
		}
		every_interest_group = {
			limit = {
				leader = {
					OR = {
						has_ideology = ideology:ideology_vanguardist
						has_ideology = ideology:ideology_communist
						has_ideology = ideology:ideology_anarchist
					}
				}
			}
			add_modifier = {
				name = communist_government_modifier
				days = normal_modifier_time
			}
		}
		add_loyalists = {
			value = medium_radicals
			strata = upper
		}
	}
}

# IG Converted by Vanguard
communism.3 = {
	type = country_event
	placement = root
	title = communism.3.t
	desc = communism.3.d
	flavor = communism.3.f

	duration = 3

	event_image = {
		video = "ep1_clandestine_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		has_technology_researched = political_agitation
		any_interest_group = {
			ig_counts_as_marginal = yes
			OR = {
				is_interest_group_type = ig_rural_folk
				is_interest_group_type = ig_intelligentsia
				is_interest_group_type = ig_armed_forces
			}
			any_scope_character = {
				NOR = {
					has_role_of_type = agitator
					has_role_of_type = ruler
				}
				has_ideology = ideology:ideology_vanguardist
			}
			leader = {
				NOR = {
					has_ideology = ideology:ideology_vanguardist
					has_ideology = ideology:ideology_communist
					has_ideology = ideology:ideology_anarchist
				}
			}
		}
	}

	immediate = {
		random_interest_group = {
			limit = {
				ig_counts_as_marginal = yes
				OR = {
					is_interest_group_type = ig_rural_folk
					is_interest_group_type = ig_intelligentsia
					is_interest_group_type = ig_armed_forces
				}
				leader = {
					NOR = {
						has_ideology = ideology:ideology_vanguardist
						has_ideology = ideology:ideology_communist
						has_ideology = ideology:ideology_anarchist
					}
				}
			}
			save_scope_as = new_vanguard_ig
			set_variable = chose_vanguardism
			random_scope_character = {
				limit = {
					NOR = {
						has_role_of_type = agitator
						has_role_of_type = ruler
					}
					has_ideology = ideology:ideology_vanguardist
				}
				save_scope_as = new_vanguard_leader
				set_as_interest_group_leader = yes
			}
		}
	}

	option = {
		name = communism.3.a
		default_option = yes
		custom_tooltip = new_vanguard.tt
		scope:new_vanguard_ig = {
			add_modifier = {
				name = new_vanguard_opinion
				days = normal_modifier_time
			}
		}
	}
	option = {
		name = communism.3.b
		custom_tooltip = new_vanguard.tt
		scope:new_vanguard_ig = {
			add_modifier = {
				name = new_vanguard_clout
				days = normal_modifier_time
			}
		}
	}
}
