#Voluntary vs forced collectivization
cre_events.1 = {
	type = country_event
	placement = root
	title = cre_events.1.t
	desc = cre_events.1.d
	flavor = cre_events.1.f
	duration = 3

	event_image = {
		video = "europenorthamerica_russian_serfs"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/invention_icons/enclosure.dds"

	trigger = {
		#triggered by enacting collectivized agriculture
	}

	#voluntary
	option = {
		name = cre_events.1.a
		default_option = yes
		show_as_tooltip = {
			add_modifier = {
				name = cre_gradual_collectivization
				years = 15
				is_decaying = yes
			}
		}
		every_interest_group = {
			limit = {
				law_stance = {
					law = law_type:law_collectivized_agriculture
					value >= approve
				}
			}
			add_modifier = {
				name = cre_collectivization_frustration
				years = 5
			}
		}
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					any_scope_politician = {
						OR = {
							is_interest_group_leader = yes
							is_ruler_of_own_country = yes
							has_role_of_type = bpm_character_role_hog
							prominence >= 50
						}
						interest_group = {
							is_in_government = yes
						}
						OR = {
							bpm_leader_is_anarachist = yes
							is_interest_group_type = ig_agrarian_populitsts
						}
					}
				}
				add = 100
			}
		}
	}

	#paid
	option = {
		name = cre_events.1.b
		hidden_effect = {
			remove_modifier = cre_gradual_collectivization
		}
		add_modifier = {
			name = cre_gradual_collectivization
			years = 7
			is_decaying = yes
		}
		add_modifier = {
			name = cre_collectivization_payments
			multiplier = money_amount_multiplier_medium
			years = 7
		}
		ai_chance = {
			base = 50
		}
	}
	#forced
	option = {
		name = cre_events.1.c
		hidden_effect = {
			remove_modifier = cre_gradual_collectivization
		}
		add_radicals = {
			value = 0.25
			pop_type = peasants
		}
		add_radicals = {
			value = 0.50
			pop_type = farmers
		}
		add_modifier = {
			name = cre_resistance_to_collectivization
			years = 5
		}
		ai_chance = {
			base = 25
			modifier = {
				trigger = {
					any_scope_politician = {
						OR = {
							is_interest_group_leader = yes
							is_ruler_of_own_country = yes
							has_role_of_type = bpm_character_role_hog
							prominence >= 50
						}
						interest_group = {
							is_in_government = yes
						}
						has_ideology = ideology:ideology_vanguardist
					}
				}
				add = 200
			}
		}
	}
}

cre_events.4 = {
	type = country_event
	placement = root
	title = cre_events.4.t
	desc = cre_events.4.d
	flavor = cre_events.4.f

	duration = 3

	icon = "gfx/interface/icons/event_icons/event_military.dds"
	cooldown = {
		days = never_fire_again_modifier_time
	}
	event_image = {
		video = "ip3_officer_inspecting_peasant_conscripts"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		NOR = {	
			has_variable = red_army_var
			has_variable = black_army_var
		}
		has_law_or_variant = law_type:law_council_republic
		has_law_or_variant = law_type:law_anarchy
		OR = {
			is_revolutionary = yes
			ig:ig_armed_forces ?= {
				leader = {
					bpm_leader_is_anarachist = yes
				}
			}
		}
		OR = {
			has_law_or_variant = law_type:law_mass_conscription
			has_law_or_variant = law_type:law_national_militia
		}
	}
	immediate = {
		ig:ig_armed_forces ?= {
			save_scope_as = army_ig
			leader = {
				save_scope_as = army_general
			}
		}
		set_variable = black_army_var
	}

	option = {
		name = cre_events.4.a
		default_option = yes
		cre_black_army_effect = yes
		# active_law:lawgroup_army_model = {
		# 	add_amendment = {
		# 		type = cre_amendment_elected_officers
		# 		sponsor = prev.ig:ig_armed_forces
		# 		cooldown = 60
		# 	}
		# }
		
		show_as_tooltip = {
			bpm_cre_ig_make_black_army = yes
		}
		custom_tooltip = {
			text = replace_commanders_with_anarchists
			every_scope_character = {
				limit = {
					popularity < 50
					OR = {
						has_role_of_type = general
						has_role_of_type = admiral
					}
					NOR = {
						has_ideology = ideology:ideology_moderate
						has_socialist_ideology = yes
					}
				}
				commander_military_formation ?= {
					save_scope_as = replacement_formation
				}
				if = {
					limit = { 
						has_role_of_type = general
					}
					root = {
						create_character = {
							is_general = yes
							interest_group = ig:ig_anarchists
							ideology = ideology:ideology_anarchist
							save_scope_as = new_commander
						}
						scope:new_commander ?= {
							transfer_to_formation = scope:replacement_formation
						}
					}
				} else = {
					root = {
						create_character = {
							is_admiral = yes
							interest_group = ig:ig_anarchists
							ideology = ideology:ideology_anarchist
							save_scope_as = new_commander
						}
						scope:new_commander ?= {
							transfer_to_formation = scope:replacement_formation
						}
					}
				}
				if = {
					limit = {
						is_historical = yes
					 }
					 exile_character_with_martyrdom = yes
				} else = {
					retire_character = yes
				}
			}
		}
		activate_law = law_type:law_elected_oficers
		add_modifier = {
			name = cre_anarchy_in_the_ranks_modifier
			years = 4
			is_decaying = yes
		}
		add_loyalists = {
			value = large_radicals
			pop_type = soldiers
		}
		add_radicals = {
			value = medium_radicals
			pop_type = officers
		}
	}
}
