on_monthly_pulse_country = {
	on_actions = {
		cre_legitimacy_action
		delay = { days = 10 }
		cre_events_action
	}
}

cre_events_action = {
	random_events = {
		chance_of_no_event = 75
		100 = cre_events.2 #revolutionary militias
		100 = cre_events.3 #women in the military
		100 = cre_events.7 #Spanish anti-clericalism
		
		#post-je_communism_2 events
		100 = cre_events.10 #Workers of the World
		100 = cre_events.11 #World Revolution
	}
	events = {
		cre_events.4 #Black Army
        sul_events.1 #Union Laws
	}
}

cre_legitimacy_action = {
	effect = {
		if = {
			limit = {
				active_law:lawgroup_distribution_of_power ?= {
					has_amendment = amendment_type:cre_amendment_anarchy_secured
				}
			 }
			 if = {
				limit = {
					has_modifier = cre_legitimacy_from_sol_modifier
				 }
				 remove_modifier = cre_legitimacy_from_sol_modifier
			 }
			 add_modifier = {
				name = cre_legitimacy_from_sol_modifier
				multiplier = {
					value = average_sol
					multiply = modifier:country_cre_legitimacy_from_sol_add
				}
			 }
		} else = {
			 if = {
				limit = {
					has_modifier = cre_legitimacy_from_sol_modifier
				 }
				 remove_modifier = cre_legitimacy_from_sol_modifier
			 }
		}
	}
}
