cre_hold_company_election = {
	icon = "gfx/interface/icons/law_icons/cooperative_ownership.dds"
	clicksound = "event:/SFX/DLC/1.3_ip1/UI/agitator_promote"

	potential = {
		is_in_exile_pool = no
		is_in_void = no
		is_ruler_of_own_country = no
		is_immortal = no
        has_role_of_type = executive
		owner ?= {
			exists = this
			has_law_or_variant = law_type:law_cooperative_ownership
		}
		is_adult = yes
	}

	possible = {
        OR = {
            popularity <= 15
            has_trait = inept
            has_trait = psychological_affliction
            law_stance = {
                law = law_type:law_cooperative_ownership
                value <= disapprove
            }
            trigger_if = {
                limit = {
                    owner ?= {
                        has_law_or_variant = law_type:law_council_republic
                    }   
                }
                law_stance = {
                    law = law_type:law_council_republic
                    value <= disapprove
                } 
            }
        }
	}
    show_requirements = yes
	cooldown = {
		years = 1
	}

	effect = {
		retire_character = yes
	}

	ai_chance = {
		value = 0
        if = {
            limit = {
                 OR = {
                    has_trait = inept
                    has_trait = psychological_affliction
                    law_stance = {
                        law = law_type:law_cooperative_ownership
                        value <= disapprove
                    }
                    trigger_if = {
                        limit = {
                            owner ?= {
                                has_law_or_variant = law_type:law_council_republic
                            }   
                        }
                        law_stance = {
                            law = law_type:law_council_republic
                            value <= disapprove
                        } 
                    }
                }  
            }
            add = 50
        }
	}
}
