REPLACE_OR_CREATE:law_state_atheism = {
	group = lawgroup_church_and_state
	
	icon = "gfx/interface/icons/law_icons/state_atheism.dds"

	progressiveness = 0

	unlocking_technologies = {
		empiricism
	}

	disallowing_laws = {
		law_theocracy
		#law_anarchy
	}

	unlocking_laws = {
		law_secret_police
		law_national_guard
	}
	
	can_enact = {
		ig:ig_devout = {
			is_in_government = no
		}
		OR = {
			is_power_bloc_leader = no
			power_bloc ?= {
				NOT = { has_identity = identity:identity_religious }
			}
		}
	}
		
	modifier = {
		state_conversion_mult = 0.25
		country_authority_add = 200

		country_acceptance_state_religion_add = 15
		country_acceptance_shared_religious_trait_add = 0
		country_acceptance_shared_religious_trait_group_add = -5
		country_acceptance_no_shared_religious_trait_add = -10

		country_allow_conversion_violent_hostility_bool = no
		country_allow_conversion_cultural_erasure_bool = yes
		country_allow_conversion_open_prejudice_bool = yes
		country_allow_conversion_second_rate_citizen_bool = yes
		country_allow_conversion_full_acceptance_bool = yes
	}

	institution = institution_home_affairs
	institution_modifier = {
		state_pop_support_movement_religious_majority_mult = -0.1
		state_pop_support_movement_religious_minority_mult = -0.1
		interest_group_ig_devout_pol_str_mult = -0.1
	}
	
	on_activate = { #PBF
		if = {
			limit = {
				NOT = { country_has_state_religion = rel:atheist }
			}
			set_state_religion = rel:atheist
		}
		add_modifier = {
			name = antireligious_policies
			days = long_modifier_time
		}
		custom_tooltip = {
			text = state_atheism_conversion_tt
			every_scope_state = {
				limit = {
					is_incorporated = yes
				}
				convert_population = {
					target = rel:atheist
					value = 0.1
				}
			}
		}
		recalculate_pop_ig_support = yes
	}

	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_church_and_state_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}

	ai_impose_chance = {
		value = 0
		
		if = {
			limit = {
				has_law = law_type:law_state_atheism
				power_bloc ?= { has_identity = identity:identity_ideological_union }			
			}
			
			add = base_impose_law_weight	
			
			# Atheist leaders of ideological unions want to spread the cult of reason
			if = {
				limit = { scope:target_country = { has_law_or_variant = law_type:law_total_separation } }
				multiply = 0.5
			}
			else = {
				multiply = 2
			}
		}

		if = {
			limit = {
				scope:target_country = {
					c:BIC ?= this
					OR = {	
						has_journal_entry = je_british_dictate_plantations
						has_journal_entry = je_british_dictate_military
						has_journal_entry = je_british_dictate_universities
					}
				}
			}
			multiply = 0
		}
	}
}

REPLACE_OR_CREATE:law_repeatable_religion_anti_religious_violence = {
	group = lawgroup_repeatable_religion
	icon = "gfx/interface/icons/law_icons/repeatable_religion_anti_religion.dds"
	
	progressiveness = 50
	
	modifier = {
		bpm_decorative_repeatable_effect_no_permanent = yes
		bpm_decorative_repeatable_effect_temporary = yes
		bpm_decorative_repeatable_stronger_from_home_affairs = yes
	}

	on_enact = {
		bpm_add_political_stability_normal = yes

		custom_tooltip = bpm_law_effect_tt_repeatable_stronger_from_home_affairs
		custom_tooltip = bpm_law_effect_tt_repeatable_anti_religious_violence_conversion
		hidden_effect = {
			every_scope_state = {
				limit = { is_incorporated = yes }
				convert_population = {
					target = rel:atheist
					value = {
						value = bpm_repeatable_law_anti_religious_violence_conversion_percent
						multiply = 0.01
					}
				}
			}
		}
		kill_population_percent = {
			percent = 0.002
			religion = root.country_definition.religion
		}
		add_modifier = {
			name = bpm_repeatable_law_anti_religious_violence_result
			years = bpm_repeatable_law_result_duration
			is_decaying = yes
			multiplier = {
				value = bpm_repeatable_law_anti_religious_violence_conversion_percent
				multiply = 0.05
			}
		}
		
		bpm_finalize_repeatable_law = {
			LAW = law_repeatable_religion_anti_religious_violence
		}
	}

	unlocking_technologies = {
		dialectics
	}

	can_enact = {
		has_institution = institution_home_affairs
		bpm_repeatable_law_not_on_cooldown = {
			LAW = law_repeatable_religion_anti_religious_violence
		}
	}

	disallowing_laws = {
		#law_anarchy
		law_no_home_affairs
		law_guaranteed_liberties
	}
	
	unlocking_laws = { 
		law_state_atheism
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { 
				has_journal_entry = je_government_petition
				has_variable = desired_law_var
                scope:law = var:desired_law_var
			}
			add = 750
		}
	}
}
