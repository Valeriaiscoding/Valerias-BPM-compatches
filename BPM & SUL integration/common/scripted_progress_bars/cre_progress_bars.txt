			desc = "BPM_CONSCIOUSNESS_LABOR"
		}

			value = ig:ig_intelligentsia.ig_approval
			divide = 10
			desc = "intelligentsia_approval"
		}
		if = {
			limit = {
				ig:ig_intelligentsia = {
					is_in_government = yes
				}
			}
			add = {
				value = 0.5
				desc = "intelligentsia_in_government"
			}
		}
		if = {
			limit = {
				ig:ig_intelligentsia = {
					leader = {
						bpm_leader_is_socialist = yes
					}
				}

			add = {
				value = ig:ig_intelligentsia.leader.popularity
				divide = 100
				desc = "socialist_intelligentsia_leader"
			}
		}
	}

 	start_value = 0
 	min_value = 0
 	max_value = 60
}

cre_undo_evolutions_bar = {
	name = "cre_undo_evolutions_bar"
 	desc = "cre_undo_evolutions_bar_desc"

 	default = yes

 	monthly_progress = {
		add = {
			value = owner.bpm_consciousness_liberal_percent
			#min = 0.5
			desc = "BPM_CONSCIOUSNESS_LIBERAL"
		}
		add = {
			value = ig:ig_intelligentsia.ig_approval
			divide = 10
			desc = "intelligentsia_approval"
		}
		if = {
			limit = {
				owner = {
					any_scope_state = {
						is_incorporated = yes
						loyalist_fraction = {
							value >= 0.10
						}
					}
				}
			}
			owner = {
				add = {
					desc = "communism_progress_from_loyalists_tt"
					every_scope_state = {
						limit = {
							is_incorporated = yes
							loyalist_fraction = {
								value >= 0.1
						}
						}
						add = this.loyalist_fraction
					}
					divide = num_incorporated_states
					multiply = 3 # Maxes out at -3
				}
			}
		}
		if = {
			limit = {
				any_political_movement = {
					bpm_movement_is_socialist = yes
				}
			}
			subtract = {
				desc = "communism_progress_from_movement_support_tt"
				every_political_movement = {
					limit = {
						bpm_movement_is_socialist = yes
					}
					add = {
						value = this.political_movement_support
						multiply = 10
					}
				}
			}
		}
		if = {
			limit = {
				any_political_movement = {
					bpm_movement_is_socialist = yes
				}
			}
			subtract = {
				desc = "communism_progress_from_movement_activism_tt"
				every_political_movement = {
					limit = {
						bpm_movement_is_socialist = yes
					}
					add = {
						value = this.political_movement_radicalism
						multiply = 5
					}
				}
			}
		}	
	}

 	start_value = 0
 	min_value = 0
 	max_value = 100
}
