REPLACE:je_anarchy = {
	icon = "gfx/interface/icons/invention_icons/anarchy.dds"

	group = je_group_cre_socialism
	should_be_pinned_by_default_uninvolved_or_context = yes
	complete = {
		average_sol >= 16
		has_law_or_variant = law_type:law_anarchy
		has_law_or_variant = law_type:law_cooperative_ownership
		has_law_or_variant = law_type:law_collectivized_agriculture
		has_law_or_variant = law_type:law_elected_police
		has_law_or_variant = law_type:law_protected_speech
		has_law_or_variant = law_type:law_elected_officers
		NOT = {
			has_law_or_variant = law_type:law_state_religion
		}
		OR = {
			has_law_or_variant = law_type:law_no_home_affairs
			has_law_or_variant = law_type:law_guaranteed_liberties
		}
		has_law_or_variant = law_type:law_slavery_banned
		has_revolution = no
	}

	on_complete = {
		trigger_event = {
			id = communism.4
			popup = yes
		}
		hidden_effect = {
			if = {
				limit = {
					any_scope_state = {
						is_incorporated = yes
						is_capital = no
						count > 1
					}
				 }
				trigger_event = {
					id = cre_events.6
					months = 6
				}
			}

		}
	}

	event_outcome_completed_effect_desc = {
		header = communism.4.a
		effect = {
			execute_event_option = { event = communism.4 option = 0 }
		 }
	}

	fail = {
		NOT = { has_law_or_variant = law_type:law_council_republic }
	}
}
