REPLACE:je_cre_communist_utopia = {
	icon = "gfx/interface/icons/invention_icons/socialism.dds"

	group = je_group_cre_socialism
	should_be_pinned_by_default_uninvolved_or_context = yes

	immediate = {
	}

	#check pop stuff on monthly for performance reasons
	on_monthly_pulse = {
		effect = {
			if = {
				limit = { 
					NOT = {
						any_scope_pop = {
							standard_of_living < 35
						}
					}
				}
				set_variable = minimum_sol_35
			} else = {
				remove_variable = minimum_sol_35
			}	
		}
	}
	complete = {
		is_at_war = no
		average_sol >= 40
		custom_tooltip = { 
			text = minimum_sol_35
			has_variable = minimum_sol_35 #This is not Omelas
		}
		literacy_rate >= 0.95
		scaled_debt < 0.25
		custom_tooltip = {
			text = no_peasants_aristocrats_capitalists_slaves
			pop_type_percent_country = {
				pop_type = peasants
				percent = 0
			}
			pop_type_percent_country = {
				pop_type = slaves
				percent = 0
			}
			pop_type_percent_country = {
				pop_type = aristocrats
				percent = 0
			}
			pop_type_percent_country = {
				pop_type = capitalists
				percent = 0
			}
		}
		NOT = {
			any_scope_state = {
				has_famine = yes
			}
		}
		country_turmoil < 0.10 
		any_scope_state = {
			filter = {
				is_incorporated = yes
			}
			percent = 1
			has_building = building_power_plant
			has_building = building_university
			has_building = building_art_academy
		}
		has_law_or_variant = law_type:law_public_schools
		institution_investment_level = {
			institution = institution_schools
			value >= 5
		}
		has_law_or_variant = law_type:law_public_health_insurance
		institution_investment_level = {
			institution = institution_health_system
			value >= 5
		}
		has_law_or_variant = law_type:law_old_age_pension
		institution_investment_level = {
			institution = institution_social_security
			value >= 5
		}
		has_law_or_variant = law_type:law_worker_protections
		has_law_or_variant = law_type:law_protected_leisure
		institution_investment_level = {
			institution = institution_workplace_safety
			value >= 5
		}
		approaching_bureaucracy_shortage = no
		OR = {
			has_law_or_variant = law_type:law_command_economy
			has_law_or_variant = law_type:law_cooperative_ownership
		}
		has_law_or_variant = law_type:law_slavery_banned
		NOR = {
			in_default = yes
			has_modifier = declared_bankruptcy
		}
	}

	on_complete = {
		trigger_event = {
			id = cre_events.9
			popup = yes
		}
	}

	event_outcome_completed_effect_desc = {
		effect = { 
			add_loyalists = {
				value = 0.75
				strata = lower
			}
			add_loyalists = {
				value = 0.35
				strata = middle
			}
			add_modifier = cre_hedonic_treadmill
		}
	}

	fail = {
		NOT = { has_law_or_variant = law_type:law_council_republic }
	}
}

				leader = {
					has_ideology = ideology:ideology_social_democrat

					has_ideology = ideology:ideology_socialist_leader

		NOT = {
			has_law_or_variant = law_type:law_advisory_body
		}

			has_law_or_variant = law_type:law_party_single
			has_law_or_variant = law_type:law_party_anti_socialist

		institution_investment_level = {
			institution = institution_schools
			value >= 5
		}
		has_law_or_variant = law_type:law_public_health_insurance
		institution_investment_level = {
			institution = institution_health_system
			value >= 5
		}
		has_law_or_variant = law_type:law_old_age_pension

		}
		has_law_or_variant = law_type:law_worker_protections
		has_law_or_variant = law_type:law_protected_leisure

			institution = institution_workplace_safety
			value >= 5
		}

				political_movement_support > 0.05

			}
		}
	}

	on_complete = {
		trigger_event = {
			id = cre_events.12
		}
	}

	event_outcome_completed_effect_desc = {
		header = cre_events.12.a
		effect = {
			execute_event_option = { event = cre_events.12 option = 0 }

		header = cre_events.12.b

			execute_event_option = { event = cre_events.12 option = 1 }

				bpm_ig_is_marginal = yes

				OR = {
					is_political_movement_type = movement_socialist
        			is_political_movement_type = movement_reformist_socialist
