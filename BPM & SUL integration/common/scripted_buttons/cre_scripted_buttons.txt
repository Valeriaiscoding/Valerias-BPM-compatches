cre_purge_academia_button = {
	name = "cre_purge_academia_button"
	desc = "cre_purge_academia_button_desc"

	visible = {
		always = yes
	}

	possible = {
		has_law_or_variant = law_type:law_public_schools
		OR = {
			has_law_or_variant = law_type:law_censorship
			has_law_or_variant = law_type:law_outlawed_dissent
		}
		OR = {
			has_law_or_variant = law_type:law_secret_police
			has_law_or_variant = law_type:law_national_guard
		}
		NOT = {
			has_law_or_variant = law_type:law_guaranteed_liberties
		}
	}

	cooldown = {
		years = 10
	}

	effect = {
		add_loyalists = {
			value = large_radicals
			interest_group = ig:ig_intelligentsia
		}
		add_radicals = {
			value = small_radicals
			pop_type = academics
		}
		kill_population_percent = {
			percent = 0.01
			pop_type = academics
		}
		every_scope_character = {
			limit = {
				has_socialist_ideology = no
				interest_group = {
					is_interest_group_type = ig_intelligentsia
				}
			}
			exile_character = yes
		}
		if = {
			limit = {
				ruler = {
					has_socialist_ideology = yes
				}
			 }
			ruler.ideology = {
				save_scope_as = party_ideology
			}
		} else = {
			ideology:ideology_communist = {
				save_scope_as = party_ideology
			}
		}
		create_character = {
			interest_group = ig:ig_intelligentsia
			ideology = scope:party_ideology
			ig_leader = yes
			on_created = {
				add_trait = political_appointee
			}
		}
		add_modifier = {
			name = cre_purged_academia
			years = 5
			is_decaying = yes
		}
	}


}
