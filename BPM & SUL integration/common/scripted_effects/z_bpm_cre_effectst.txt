}

REPLACE:cre_devolve_commune_effect = {
	random_scope_state = {
		limit = {
			is_incorporated = yes
			is_capital = no
		}
		save_scope_as = new_commune_state
	}
	create_dynamic_country = {
		origin = root
		country_type = recognized
		tier = principality
		capital = scope:new_commune_state
		on_created = {
			activate_law = law_type:law_anarchy
			activate_law = law_type:law_confederated
			create_political_movement = {
				type = movement_anarchist
			}
			set_ruling_interest_groups = {
				ig_anarchists
			}
			ruler ?= {
				set_interest_group = ig:ig_anarchists
				set_ideology = ideology:ideology_anarchist
			}
			ig:ig_trade_unions ?= {
				leader = {
					set_ideology = ideology:ideology_anarchist
				}
			}
			create_bidirectional_truce = {
				country = root
				months = 60
			}
			add_modifier = {
				name = cre_anarchist_center_modifier
				years = 25
			}
		}
	}
	hidden_effect = {
		create_diplomatic_pact = {
			country = scope:new_commune_state.owner
			type = puppet
		}
		create_diplomatic_pact = {
			country = scope:new_commune_state.owner
			type = da_overlord_grant_investment_rights
		}
		create_diplomatic_pact = {
			country = scope:new_commune_state.owner
			type = decrease_payments
		}
		scope:new_commune_state.owner = {
			add_liberty_desire = -100
			change_relations = {
				country = root
				value = 200
			}
		}
		if = {
			limit = {
				is_in_power_bloc = yes
			 }
			 power_bloc = {
				add_leverage = {
					target = scope:new_commune_state.owner
					value = 2000
				}
			 }
		}
	}
}

cre_pb_to_managers_effect = {
	ig:ig_petty_bourgeoisie ?= {
		set_interest_group_name = ig_managers
		hidden_effect = {
			if = {
				limit = {
					has_ideology = ideology:ideology_reactionary
				}
				remove_ideology = ideology_reactionary
			}
			if = {
				limit = {
					has_ideology = ideology:ideology_modernizer
				}
				remove_ideology = ideology_modernizer
			}
			if = {
				limit = {
					has_ideology = ideology:ideology_republican
				}
				remove_ideology = ideology_republican
			}
		}
		usfp_remove_american_ideologies = yes
		if = {
			limit = {
				owner = {
					has_law_or_variant = law_type:law_cooperative_ownership
				}
			 }
			add_ideology = ideology_eco_proletarian_economy
		} 
		else = {
			add_ideology = ideology_eco_socialist_autarky
		}
		bpm_reorder_ideologies = yes
	}
}

REPLACE:bpm_rename_trade_unions_ig = {
	if = {
		limit = {
			c:USA ?= THIS
			OR = {
				has_variable = bpm_2nd_party_system
				game_date <= 1836.1.1
			}
		}
		ig:ig_trade_unions ?= { set_interest_group_name = ig_locofocos }
	}
	else_if = {
		limit = {
			has_law_or_variant = law_type:law_combination_acts
		}
		ig:ig_trade_unions ?= { set_interest_group_name = ig_labor_radicals }
	}
	else_if = {
		limit = { ig:ig_trade_unions ?= { has_ideology = ideology:ideology_dop_syndical_unionist } }
		ig:ig_trade_unions ?= { set_interest_group_name = ig_syndicalists }
	}
	else_if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_council_republic
				has_law_or_variant = law_type:law_factory_councils
			}
		}
		set_interest_group_name = ig_workers_councils
	}
	else_if = {
		limit = { has_variable = bpm_solidarity_knights_of_labor }
		ig:ig_trade_unions ?= { set_interest_group_name = ig_knights_of_labor }
	}
	else_if = {
		limit = {
			has_law_or_variant = law_type:law_right_to_associate
			NOT = { has_variable = union_law_introduced_var }
		}
		ig:ig_trade_unions ?= { set_interest_group_name = ig_urban_workers }
	}
	else_if = {
		limit = { has_variable = bpm_solidarity_national_grange }
		ig:ig_trade_unions ?= { set_interest_group_name = ig_national_grange }
	}
	else_if = {
		limit = {
			ig:ig_trade_unions = {
				has_variable = bpm_conservative_unions
			}
		}
		ig:ig_trade_unions ?= { set_interest_group_name = ig_conservative_unions }
	}
	else = {
		ig:ig_trade_unions ?= { set_interest_group_name = ig_trade_unions }
	}
}
