REPLACE_OR_CREATE:law_command_economy = {
	group = lawgroup_economic_system
	
	icon = "gfx/interface/icons/law_icons/command_economy.dds"
	
	progressiveness = 100

	is_visible = {
		OR = {
			AND = {
				com_law_command_economy_trigger = yes
				# Any vanilla conditions that have been added
				always = yes
			}
			com_law_command_economy_alternative_trigger = yes
		}
	}
	
	disallowing_laws = {
		law_serfdom
		law_manorialism
		law_bakuhan_system
		law_anarchy
		law_women_in_the_fields
	}

	
	unlocking_technologies = {
		central_planning
	}
	
	unlocking_laws = {
		#law_council_republic
		#law_technocracy
	}	

	can_enact = {
		NOT = {
			active_law:lawgroup_economic_system = {
				has_amendment = amendment_type:bpm_amendment_artisanal_economy
			}
		}
		OR = {
			custom_tooltip = {
				text = "christdems_have_centralized_economy_tooltip"
				OR = {
					owner = {
						OR = {
							ig:ig_christian_democrats ?= {
								has_ideology = ideology_eco_centralized_economy
								is_in_government = yes
							}
							ig:ig_christian_democrats = {
								has_ideology = ideology_eco_centralized_economy
								is_in_government = yes
							}
						}
					}
					OR = {
						ig:ig_christian_democrats ?= {
							has_ideology = ideology_eco_centralized_economy
							is_in_government = yes
						}
						ig:ig_christian_democrats = {
							has_ideology = ideology_eco_centralized_economy
							is_in_government = yes
						}
					}
				}
			}
			has_law = law_type:law_council_republic
			has_law = law_type:law_technocracy
		}
	}
	
	on_activate = {
		seize_investment_pool = yes
	}
	
	modifier = {
		country_disable_investment_pool_bool = yes
		
		building_nationalization_cost_mult = -0.5
		building_nationalization_radicals_mult = -0.75
		country_institution_economy_max_investment_add = 4
		
		country_leverage_resistance_mult = 0.25

		state_pop_pol_str_add = -4

		country_disable_non_company_privatization_bool = yes
	}

	institution = institution_economy
	institution_modifier = {
        state_construction_mult = 0.25
		country_authority_mult = 0.05
		state_bureaucrats_investment_pool_efficiency_mult = 0.08
		country_bureaucrats_pol_str_mult = 0.05
		country_government_dividends_efficiency_add = 0.2
		state_pop_pol_str_add = 0.5

		building_company_government_dividends_add = 0.15
	}
	
	ai_will_do = {
		OR = {
			has_law = law_type:law_technocracy
			AND = {
				has_law = law_type:law_council_republic
				ruler ?= {
					has_ideology = ideology:ideology_vanguardist
					has_ideology = ideology:ideology_communist
				}			
			}
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
		
		if = {
			limit = { 
				has_law = law_type:law_council_republic
			}
			add = 500
		}
		
		if = {
			limit = { 
				has_law = law_type:law_technocracy
			}
			add = 100
		}
	}
}
