bpm_ecchi_socialists_split_set_ig_leader = {
	hidden_effect = {
		ordered_scope_character = {
			limit = {
				prominence >= 25
				is_interest_group_leader = no
			}
			order_by = prominence
			position = 0
			check_range_bounds = no
			if = { 
				limit = { NOT = { has_role_of_type = politician } }
				add_character_role = politician
			}
			if = {
				limit = { has_role_of_type = agitator }
				remove_character_role = agitator
			}
			set_as_interest_group_leader = yes
		}
	}
}
