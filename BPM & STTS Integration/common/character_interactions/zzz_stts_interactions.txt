		is_in_void = no
		is_ruler_of_own_country = no
		is_immortal = no
		OR = {
			is_heir_of_own_country = yes
			has_role_of_type = general
			has_role_of_type = admiral
			has_role_of_type = politician
			has_role_of_type = executive
		}

			exists = this
			has_law = law_type:law_single_legal_party
		}
		is_adult = yes
		trigger_if = {
			limit = { has_law = law_type:law_supreme_legislature }
			NOT = {
				owner.var:bpm_headofgov.ideology = this
			}
			NOT = {
				ideology = owner.var:bpm_headofgov.ideology
			}
		}
		trigger_else_if = {
			limit = { bpm_has_gov_regency = yes }
			NOT = {
				owner.var:bpm_regent = this
			}
			NOT = {
				ideology = owner.var:bpm_regent.ideology
			}
		}
		trigger_else = {
			NOT = {
				ideology = owner.ruler.ideology
			}
		}
		
		# trigger_if = {
		# 	limit = { has_law = law_type:law_no_parties_allowed }
		# 	NOT = {
		# 		law_stance = {
		# 			law = law_type:law_organic_democracry
		# 			value >= approve
		# 		}
		# 	}
		# }
		# trigger_else = {
		# 	NOT = {
		# 		ideology = owner.ruler.ideology
		# 	}
		# }
	}

	possible = {
		owner ?= {
			has_institution = institution_home_affairs
			NOR = {
				has_law_or_variant = law_type:law_guaranteed_liberties
				has_law_or_variant = law_type:law_protected_speech
			}
			government_legitimacy >= 25
		}
		OR = {
			interest_group ?= {
				is_in_government = yes
			}
			AND = {
				has_role_of_type = executive
				owner ?= {
					has_law_or_variant = law_type:law_command_economy
				} 
			}
		}
	}

	cooldown = {
		years = 3
	}

	show_requirements = yes

	effect = {
		save_scope_as = relevant_character
		owner ?= {
			add_modifier = {
				name = stts_enforcing_party_ideology_modifier
				years = 3
			}
		}
		if = {
			limit = {
				exists = owner.institution:institution_home_affairs
			}
			random_list = {
				#success
				0 = {
					#chance based on censorship laws
					modifier = {
						if = {
							limit = { 
								owner ?= {
									has_law_or_variant = law_type:law_outlawed_dissent
								}
							}
							add = 200
						}
						else_if = {
							limit = { 
								owner ?= {
									has_law_or_variant = law_type:law_censorship
								}
							}
							add = 100
						}
						else_if = {
							limit = { 
								owner ?= {
									has_law_or_variant = law_type:law_right_of_assembly
								}
							}
							add = 50
						}
						#secret police makes this a lot easier
						if = {
							limit = {
								owner ?= {
									has_law_or_variant = law_type:law_secret_police
								}
								owner.institution:institution_home_affairs.investment >= 1
							}
							add = {
								value = 30
								multiply = owner.institution:institution_home_affairs.investment
							}
						}
					}
					if = {
						limit = {
							OR = {
								is_historical = no
								has_role_of_type = executive
							}
							#particularly committed ideologies won't toe the line
							NOR = {
								has_ideology = ideology:ideology_radical
								has_ideology = ideology:ideology_anarchist
								has_ideology = ideology:ideology_vanguardist
								has_ideology = ideology:ideology_fascist
								has_ideology = ideology:ideology_integralist
								has_trait = imperious
							}
						 }
						set_ideology = owner.ruler.ideology
						if = {
							limit = {
								NOT = {
									has_trait = compliant
								}
							}
							add_trait = compliant
						}
					} else = {
						interest_group = {
							save_scope_as = replacement_ig
						}
						owner ?= {
							create_character = {
								ideology = owner.ruler.ideology
								interest_group = scope:replacement_ig
								ig_leader = yes
								on_created = {
									add_trait = compliant
								}
							}
						}
						exile_character = yes
					}
					owner ?= {
						add_radicals = {
							value = small_radicals
							interest_group = prev.interest_group
						}
						post_notification = stts_party_discipline_success
					}
				}
				#failure
				0 = {
					#chance based on popularity and IG support
					modifier = {
						add = popularity
						add = {
							value = interest_group.ig_clout
							multiply = 10
						}
						subtract = {
							value = interest_group.ig_approval
							multiply = 10
						}
						if = {
							limit = {
								has_trait = imperious
							 }
							add = 50
						}
					}
					interest_group ?= {
						add_modifier = {
							name = stts_ig_resisting_party_discipline_modifier
							years = 3
						}
					}
					owner ?= {
						add_radicals = {
							value = large_radicals
							interest_group = prev.interest_group
						}
						post_notification = stts_party_discipline_failure	
					}
				}
			}
		}

	}

	ai_chance = {
		value = 0
	}
}
