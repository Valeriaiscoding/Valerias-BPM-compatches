	}

	interest_group_leader_trigger = {
		bpm_interest_group_is_activated = yes
		owner = { has_technology_researched = labor_movement }
		OR = {
			bpm_ig_is_middle_institutional = yes
			bpm_ig_is_lower_institutional = yes
			AND = {
				bpm_ig_is_left_wing = yes
				bpm_ig_is_radical_socialist = no
			}

		NOT = { is_interest_group_type = ig_market_liberals }
		bpm_ig_is_state_institutional = no
		OR = {
			bpm_ig_is_labor = yes
			bpm_ig_is_radical_liberal = yes
			owner = {
				any_interest_group = {
					bpm_ig_is_labor = yes
					ig_clout > 0.05
				}
			}
		}
	}
	
	interest_group_leader_weight = {
		value = 200
		# less likely for moderate liberals and committed socialists
		if = {
			limit = {
				OR = {
					bpm_ig_is_moderate_liberal = yes
					bpm_ig_is_proletarian_socialist = yes
				}
			}
			add = -75
		}
		# very angry IG's more likely to turn to more radical socialism
		if = {
			limit = {
				ig_approval <= -5
			}
			add = -75
		}
		# more likely when issues are relevant
		if = {
			limit = {
				owner = {
					bpm_country_laws_have_max_welfare = no
				}
			}
			add = 100
		} 
		else = {
			add = -100
		}
		# more likely for content devout in council republic
		if = {
			limit = {
				is_interest_group_type = ig_devout
				ig_approval >= 0
				owner = {
					has_law = law_type:law_council_republic
				}
			}
			add = 75
		}
		# more likely for content trade unions outside of council republic
		if = {
			limit = {
				is_interest_group_type = ig_trade_unions
				ig_approval >= 0
				owner = {
					NOT = { has_law = law_type:law_council_republic }
				}
			}
			add = 75
		}
		# liberal and/or radical leaders have a higher spawn chance if the Dreyfusards won.
		if = {
			limit = {
				has_variable = dreyfusard_victory
			}
			add = 10
		}
		# Pedro JE victory
		if = {
			limit = {
				owner = {
					has_variable = brz_enduring_empire
				}
			}
			add = 50
		}
		# National Liberation Alliance
		if = {
			limit = {
				owner = {
					has_variable = national_liberation_alliance_variable
					has_journal_entry = je_new_republic
				}
			}
			multiply = 1.20
		}
	}

	non_interest_group_leader_trigger = {
        interest_group = {
    		bpm_interest_group_is_activated = yes
    		owner = { has_technology_researched = labor_movement }
    		OR = {

    			bpm_ig_is_radical_liberal = yes

    				any_interest_group = {
    					bpm_ig_is_labor = yes

    	}

		NAND = {
			has_role = agitator
			owner = {
				bpm_country_laws_have_max_welfare = yes
			}
		}
	}
	
	non_interest_group_leader_weight = {
		value = 50
		interest_group = {
			if = {
				limit = {
					OR = {
						bpm_ig_is_moderate_liberal = yes
						bpm_ig_is_proletarian_socialist = yes
					}
				}
				add = -75

			# very angry IG's more likely to turn to more radical socialism
