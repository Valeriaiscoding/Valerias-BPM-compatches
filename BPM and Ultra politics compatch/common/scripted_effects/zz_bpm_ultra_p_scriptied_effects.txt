REPLACE:bpm_initialize_radlibs = {
		if = {
			limit = { 
				ig:ig_radicals ?= { 
					has_ideology = ideology:ideology_nonexistent
				}
			}
			if = {
				limit = {
					this = c:JAP
				}
				ig:ig_radicals ?= {
					remove_ideology = ideology_nonexistent
					add_ideology = ideology_gov_moderate_monarchist
				}
			}
			else = {
				ig:ig_radicals ?= {
					remove_ideology = ideology_nonexistent
					add_ideology = ideology_gov_radical_republican
				}
			}
			if = {
				limit = { 
					OR = {
						bpm_country_is_in_new_world = yes
						this ?= c:GBR
					}
				}
				ig:ig_radicals ?= { add_ideology = ideology_mil_patriotic_liberal }
			}
			#else_if = {
			#	limit = { bpm_country_is_latin_american = yes }
			#	ig:ig_radicals ?= { add_ideology = ideology_mil_antimilitarist }
			#}
			else = { ig:ig_radicals ?= { add_ideology = ideology_mil_mass_army } }

			ig:ig_radicals ?= {
				add_ideology = ideology_dop_radical_democrat
				add_ideology = ideology_sta_progressive_state
				add_ideology = ideology_cit_liberal_nationalist
				add_ideology = ideology_rel_radical_secularist
				#add_ideology = ideology_mil_patriotic_liberal
				add_ideology = ideology_eco_modernizer
				add_ideology = ideology_wel_public_welfare
				add_ideology = ideology_fem_gender_egalitarian
				add_ideology = ideology_ban_averse
				# Vanilla ideologies
				add_ideology = ideology_liberal
				add_ideology = ideology_populist
				add_ideology = ideology_republican
				add_ideology = ideology_egalitarian
				add_ideology = ideology_anti_slavery

				every_scope_character = {
					limit = {
						has_ideology = ideology:ideology_placeholder_leader
					}
					set_ideology = ideology:ideology_center_leader
				}

				bpm_reorder_ideologies = yes
			}
			bpm_rename_igs = yes
		}
}

REPLACE:bpm_setup_country_wel_ideologies = {
	ig:ig_industrialists = {
		if = {
			limit = { 
				owner = { 
					OR = {
						NOT = { has_technology_researched = corporatism } 
						country_is_colonial_or_company = yes
					}
				}
			}
			bpm_remove_all_wel_ideologies = yes
			add_ideology = ideology_wel_market_welfare
		}
		else = {
			bpm_remove_all_wel_ideologies = yes
			add_ideology = ideology_wel_paternalistic_welfare
		}
		
	}
	ig:ig_intelligentsia = {
		if = {
			limit = { has_ideology = ideology:ideology_eco_placeholder }
			bpm_remove_all_eco_ideologies = yes
			add_ideology = ideology_eco_artisan
			
		}
		bpm_remove_all_wel_ideologies = yes
		add_ideology = ideology_wel_secular_anti_welfare
	}
	if = {
		limit = {
			OR = {
				country_has_primary_culture = cu:yankee
				country_has_primary_culture = cu:dixie
				country_has_primary_culture = cu:north_german
				country_has_primary_culture = cu:alemannic
				country_has_primary_culture = cu:swedish
				country_has_primary_culture = cu:norwegian
				country_has_primary_culture = cu:danish
				country_has_primary_culture = cu:finnish
				country_has_primary_culture = cu:boer
				this = c:USA
				this = c:TEX
				this = c:LIB
				this = c:WUR
			}
		}
		ig:ig_devout = {
			bpm_remove_all_wel_ideologies = yes
			add_ideology = ideology_wel_paternalistic_welfare
		}
	}

	if = {
		limit = { bpm_country_is_in_new_world = yes }
		if = {
			limit = { country_has_state_religion = rel:catholic }
			ig:ig_reactionaries ?= { 
				if = {
					limit = { has_ideology = ideology:ideology_wel_no_welfare }
				}
				remove_ideology = ideology_wel_no_welfare
				add_ideology = ideology_wel_religious_welfare
			}
		}
	}
}
