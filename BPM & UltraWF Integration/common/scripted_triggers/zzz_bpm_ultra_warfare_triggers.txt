bpm_is_antagonistic_conscription = {
    trigger_if = {
        limit = {
            interest_group = {
                has_ideology = ideology:ideology_mil_antimilitarist
            }
        }
        always = yes
    }
    trigger_else_if = {
        limit = {
            owner = {
                has_law_or_variant = law_type:law_council_republic
            }
        }
        interest_group = {
            bpm_ig_is_upper_institutional = yes
        }
    }
    trigger_else = {
        interest_group = {
            bpm_ig_is_radical_socialist = yes
        }
    }
    
}
