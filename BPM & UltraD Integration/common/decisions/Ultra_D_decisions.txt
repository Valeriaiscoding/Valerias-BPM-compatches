break_international_treaties = {
	is_shown = {
		NOT = { is_country_type = decentralized }
		NOT = { is_country_type = unrecognized }
		NOT = { infamy >= infamy_threshold:pariah }
		is_subject = no
		aggressive_diplomatic_plays_permitted = yes
		bpm_aggressive_diplomacy_permitted = yes
	}

	possible = {
		OR = {
			ruler = { bpm_leader_is_hawkish = yes }
			ruler = { has_ideology = ideology:ideology_vanguardist }
			ruler = { has_ideology = ideology:ideology_bonapartist }
			any_interest_group = {
				is_in_government = yes
				bpm_ig_is_very_hawkish = yes
				OR = {
					is_powerful = yes
					is_strongest_ig_in_government = yes
				}
			}
			has_law = law_type:law_ultramilitarism
		}
		trigger_if = {
			limit = { has_law_or_variant = law_type:law_answers_to_legislature }
			var:bpm_headofgov ?= {
				bpm_leader_is_doveish = no
				interest_group = {
					bpm_ig_is_doveish = no
				}
			}
		}
		trigger_else = {
			ruler ?= { 
				bpm_leader_is_doveish = no
				interest_group = {
					bpm_ig_is_doveish = no
				}
			}
		}
		
	}

	when_taken = { #goal is to become international pariah so you can ignore all diplomatic considerations
		if = {
			limit = {
				infamy >= infamy_threshold:notorious
			}
			change_infamy = 55
		}
		else_if = {
			limit = {
				infamy >= infamy_threshold:infamous
			}
			change_infamy = 80
		}
		else = {
			change_infamy = 105
		}
	}

	ai_chance = {
		value = 0
		# if = {
		# 	limit = { 
		# 		has_law = law_type:law_ultramilitarism 
		# 	}
		# 	add = 100
		# }
	}
}
