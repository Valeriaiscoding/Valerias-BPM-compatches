REPLACE:movement_german_national = {
	category = movement_category_pan_national

	ideology = ideology_national_liberal_movement
	character_ideologies = {
		ideology_ethno_nationalist
		ideology_market_liberal
		ideology_liberal_leader
		ideology_republican_leader
		ideology_radical
	}

	can_cause_obstinance = yes

	creation_trigger = {
		OR = {
			country_has_primary_culture = cu:north_german
			country_has_primary_culture = cu:south_german
			country_has_primary_culture = cu:low_saxon
			country_has_primary_culture = cu:pomeranian
			country_has_primary_culture = cu:prussian
			country_has_primary_culture = cu:silesian
			country_has_primary_culture = cu:saxon
			country_has_primary_culture = cu:westphalian
			country_has_primary_culture = cu:hessian
			country_has_primary_culture = cu:rhenish
			country_has_primary_culture = cu:swabian
			country_has_primary_culture = cu:alsacian
		}
		OR = {
			cu:north_german.culture_current_fervor > 10
			cu:south_german.culture_current_fervor > 10
			cu:low_saxon.culture_current_fervor > 10
			cu:pomeranian.culture_current_fervor > 10
			cu:prussian.culture_current_fervor > 10
			cu:silesian.culture_current_fervor > 10
			cu:saxon.culture_current_fervor > 10
			cu:westphalian.culture_current_fervor > 10
			cu:hessian.culture_current_fervor > 10
			cu:rhenish.culture_current_fervor > 10
			cu:swabian.culture_current_fervor > 10
			cu:alsacian.culture_current_fervor > 10
		}
		NOT = {
			c:GER ?= this
		}
	}
	creation_weight = {
		value = 100
		if = {
			limit = {
				owner ?= { # If a movement of this type exists in a neighbouring country, make it more likely to spawn here as well
					any_neighbouring_state = {
						owner = {
							prestige > ROOT.owner.prestige
							any_political_movement = {
								is_political_movement_type = movement_german_national
							}
						}
					}
				}
			}
			multiply = {
				value = 2
			}
		}
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}

	disband_trigger = {
		c:GER ?= this
	}

	on_disbanded = {
		if = {
			limit = {
				c:GER ?= owner
			}
			owner ?= {
				create_political_movement = { type = movement_cultural_majority }
			}
		}
	}

	character_support_trigger = {
		culture = {
			has_discrimination_trait = tradition_german
		}
	}
	character_support_weight = {
		value = 100
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_ethno_nationalist
					has_ideology = ideology:ideology_fascist
					has_liberal_ideology = yes
					has_progressive_ideology = yes
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				has_conservative_ideology = yes
			}
			multiply = {
				value = 0.5
			}
		}
		else_if = {
			limit = {
				has_reactionary_ideology = yes
				NOR = {
					has_ideology = ideology:ideology_ethno_nationalist
					has_ideology = ideology:ideology_fascist
				}
			}
			multiply = {
				value = 0
			}
		}
	}

	pop_support_trigger = {
		culture = {
			has_discrimination_trait = tradition_german
		}
	}

	pop_support_factors = {
		movement_support_national_awakening
		movement_support_cultural_fervor
		movement_support_high_urbanization
		movement_support_tradition_german
		movement_support_academics
		movement_support_clerks
		movement_support_shopkeepers
		movement_support_officers
		movement_support_soldiers
		movement_support_nationalism_tech
		movement_support_pan_nationalism_tech
	}

	pop_support_weight = {
		if = {
			limit = {
				strata = upper
			}
			if = {
				limit = {
					is_pop_type = capitalists
				}
				add = {
					value = 10
					desc = "POP_CAPITALISTS"
				}
			}
			else = {
				add = {
					value = 4
					desc = "UPPER_NO_ICON"
				}
			}
		}
		else_if = {
			limit = {
				strata = middle
			}
			if = {
				limit = {
					is_pop_type = academics
				}
				add = {
					value = 35
					desc = "POP_ACADEMICS"
				}
			}
			if = {
				limit = {
					is_pop_type = shopkeepers
				}
				add = {
					value = 25
					desc = "POP_SHOPKEEPERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = officers
				}
				add = {
					value = 15
					desc = "POP_OFFICERS"
				}
			}
			else = {
				add = {
					value = 12
					desc = "MIDDLE_NO_ICON"
				}
			}
		}
		else_if = {
			limit = {
				strata = lower
			}
			if = {
				limit = {
					is_pop_type = clerks
				}
				add = {
					value = 20
					desc = "POP_CLERKS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = soldiers
				}
				add = {
					value = 10
					desc = "POP_SOLDIERS"
				}
			}
			else = {
				add = {
					value = 5
					desc = "LOWER_NO_ICON"
				}
			}
		}

		add = {
			desc = "URBAN_STATE"
			value = 10
			multiply = state.state_urbanization_rate
		}

		if = {
			limit = {
				owner ?= {
					has_technology_researched = pan-nationalism
				}
			}
			multiply = {
				value = 1.5
				desc = "PAN_NATIONALISM_TECH"
			}
		}
		else_if = {
			limit = {
				owner ?= {
					has_technology_researched = nationalism
				}
			}
			multiply = {
				value = 1.25
				desc = "NATIONALISM_TECH"
			}
		}

		multiply = { # Range from x0.775 to 2.275
			desc = "POP_CULTURE_FERVOR"
			value = culture.culture_current_fervor
			subtract = 15
			multiply = 0.02
			add = 1
		}

		if = {
			limit = {
				culture ?= {
					culture_has_national_awakening = yes
				}
			}
			multiply = {
				value = 1.5
				desc = "POP_ACTIVE_NATIONAL_AWAKENING"
			}
		}

		if = { # If a more radical movement exists, it'll poach radicals from this one.
			limit = {
				owner ?= {
					any_political_movement = {
						OR = {
							is_political_movement_type = movement_radical
							is_political_movement_type = movement_fascist
						}
					}
				}
			}
			multiply = {
				desc = "POP_RADICAL_MOVEMENT_LOSSES"
				value = 1
				subtract = pop_radical_fraction
			}
		}

		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"

			value = literacy_rate

			multiply = literacy_rate
			add = 0.10
		}
		if = {
			limit = {
				owner ?= {
					any_political_movement = {
						is_political_movement_type = movement_german_national
						OR = {
							is_being_suppressed = yes
							is_being_bolstered = yes
						}
						save_temporary_scope_as = affected_movement
					}
				}
			}
			if = {
				limit = {
					scope:affected_movement = {
						is_being_suppressed = yes
					}
				}
				multiply = {
					value = 1
					subtract = owner.modifier:country_suppression_attraction_factor
					desc = "concept_suppressed"
				}
			}
			else = {
				multiply = {
					value = 1
					add = owner.modifier:country_bolster_attraction_factor
					desc = "concept_bolstered"
				}
			}
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = 1
			if = {
			   limit = {
					NOT = {
						state_is_homeland_of_german_cultures = yes
					}
				}
				multiply = 0
			}
		}
	}

	secession = {
		possible = {
			owner ?= {
				has_law_or_variant = law_type:law_subjecthood
			}
			political_movement_support >= 0.05
			scope:clout <= 0
		}

		weight = {
			value = 1000
		}

		state_weight = {
			value = 1
			if = {
			   limit = {
					NOT = {
						state_is_homeland_of_german_cultures = yes
					}
				}
				multiply = 0
			}
		}

			NGF = {
				trigger = {
					owner ?= {
						country_has_primary_culture = cu:north_german
					}
				}
				weight = 1000
			}
			SGF = {
				trigger = {
					owner ?= {
						country_has_primary_culture = cu:south_german
					}
				}
				weight = 1000
			}
		}
	}

	additional_radicalism_factors = {
		add = { value = bpm_additional_radicalism_factors desc = "bpm_additional_radicalism_factors" }
		if = {
			limit = {
				owner.ruler ?= {
					NOT = { culture = { is_primary_culture_of = root.owner } }
				}
			}

			add = {
				value = 0.1
				desc = "RULER_NOT_PRIMARY_CULTURE"
			}
		}
		if = {
			limit = {
				owner = {
					has_journal_entry = je_schleswig_holstein_question
				}
			}
			add = {
				desc = "UNSOLVED_SCHLESWIG_HOLSTEIN_QUESTION"
				value = 0.10
			}
		}

		if = {
			limit = {
				owner ?= {
					has_technology_researched = nationalism
				}
			}
			if = {
				limit = {
					owner ?= {
						is_unification_candidate = GER
						any_diplomatically_relevant_country = {
							is_unification_candidate = GER
						}
					}
				}
				add = {
					desc = "RIVAL_GERMAN_UNIFIERS"
					value = 0.10
				}
			}
			else_if = {
				limit = {
					owner ?= {
						NOT = {
							is_unification_candidate = GER
						}
						any_diplomatically_relevant_country = {
							is_unification_candidate = GER
						}
					}
				}
				add = {
					desc = "SUPPORT_FOR_UNIFIER"
					value = 0.10
				}
			}
		}
		if = {
			limit = {
				owner ?= {
					has_technology_researched = pan-nationalism
				}
			}
			if = {
				limit = {
					owner ?= {
						is_unification_candidate = GER
						NOT = {
							any_state_region = {
								OR = {
									is_homeland = cu:south_german
									is_homeland = cu:north_german
									is_homeland = cu:low_saxon
									is_homeland = cu:pomeranian
									is_homeland = cu:prussian
									is_homeland = cu:silesian
									is_homeland = cu:saxon
									is_homeland = cu:hessian
									is_homeland = cu:rhenish
									is_homeland = cu:alsacian
									is_homeland = cu:swabian
								}
								any_scope_state = {
									NOT = {
										owner = ROOT.owner
									}
								}
							}
						}
					}
				}
				add = {
					desc = "GERMAN_UNIFICATION"
					value = 0.10
				}
			}
		}

		# if = {
		# 	limit = {
		# 		owner = {
		# 			any_neighbouring_state = {
		# 				owner = {
		# 					NOT = { THIS = ROOT.owner }

		# 	add = {
		# 		desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
		# 		value = 0.15
		# 	}
		# }

		add = {
			desc = "FERVOR_OF_GERMAN_CULTURE_ACTIVISM"
			value = owner.average_german_fervor
			divide = 250
		}
	}
