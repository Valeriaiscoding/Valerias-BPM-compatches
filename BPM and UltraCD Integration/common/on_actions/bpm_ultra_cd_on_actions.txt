on_monthly_pulse_country ={
	on_actions = {
		#bpm_ultra_cd_switch_yakubian_law
	}
}

on_country_formed = {
	on_actions = {
		bpm_ultra_cd_compat_ideology
	}
}

bpm_ultra_cd_compat_ideology = {
	effect = {
		bmp_ultra_cd_conservatives_ideology_effect = yes
	}
}

bpm_ultra_cd_switch_yakubian_law = {
	effect = {
		if = {
			limit = { 
				has_law = law_type:law_white_universalism 
				any_primary_culture = {
					OR = {
						NOT = { has_discrimination_trait_group = heritage_group_european }
						has_discrimination_trait = heritage_latin_american_settler
					}
				}
			}
			activate_law = law_type:law_cultural_exclusion
		}
		if = {
			limit = { 
				has_law = law_type:law_cultural_exclusion
				any_primary_culture = { has_discrimination_trait_group = heritage_group_european }
				NOT = {
					any_primary_culture = {
						OR = {
							NOT = { has_discrimination_trait_group = heritage_group_european }
							has_discrimination_trait = heritage_latin_american_settler
						}
					}
				}
			}
			activate_law = law_type:law_white_universalism
		}
	}
}
