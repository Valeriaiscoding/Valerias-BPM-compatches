warispolitics_capitulate = {
	effect = {
		if = {
			limit = { has_law_or_variant = law_type:law_answers_to_legislature }
			var:bpm_headofgov = {
				save_scope_as = ruler_char
			}
		}
		else = {
			ruler = {
				save_scope_as = ruler_char
			}
		}
		var:bpm_headofgov ?= {
			save_scope_as = hog
		}
		if = {
			limit = { 
				has_modifier = warispolitics_lost_war
			}
			remove_modifier = warispolitics_lost_war
		}
		add_modifier = {
			name = warispolitics_lost_war
			days = 1825
			is_decaying = yes
		}
		#IGs dinged
		every_interest_group = {
			limit = {
				OR = {
					is_in_government = yes
					is_same_interest_group_type = scope:ruler_char.interest_group
					is_same_interest_group_type = scope:hog.interest_group					
				}
			}
			if = {
				limit = { 
					has_modifier = warispolitics_ig_war_loser
				}
				remove_modifier = warispolitics_ig_war_loser
			}
			add_modifier = {
				name = warispolitics_ig_war_loser
				is_decaying = yes
				days = 1825
			}
		}
		#Ruler dinged
		scope:ruler_char ?= {
			if = {
				limit = {
					has_modifier = warispolitics_character_lost_war
				}
				remove_modifier = warispolitics_character_lost_war
			}
			if = {
				limit = { 
					OR = {
						is_ruler_of_own_country = no 
						owner = { bpm_is_hoshog = yes }
					}
				}
				add_modifier = {
					name = warispolitics_character_lost_war
					is_decaying = yes
					multiplier = 2
					days = 1825
				}
			}
			else = {
				add_modifier = {
					name = warispolitics_character_lost_war
					is_decaying = yes
					days = 1825
				}
			}
		}
		if = {
			limit = { bpm_is_hoshog = no }
			scope:hog ?= {
				if = {
					limit = {
						has_modifier = warispolitics_character_lost_war
					}
					remove_modifier = warispolitics_character_lost_war
				}
				add_modifier = {
					name = warispolitics_character_lost_war
					is_decaying = yes
					days = 1825
				}
			}
		}
		
	}
}
