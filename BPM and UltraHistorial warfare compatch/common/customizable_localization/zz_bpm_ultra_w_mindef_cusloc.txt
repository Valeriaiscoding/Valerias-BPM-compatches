bpm_mindef_name = {
	type = country
	random_valid = no
	text = {
		trigger = {
			bpm_shogunate_trigger = yes
		}
		localization_key = bpm_mindef_name_shogunate
	}
	text = {
		trigger = {
			exists = c:USA
			THIS ?= c:USA
			bpm_country_is_bourgeois_republic = yes
			OR = {
				has_law_or_variant = law_type:law_justum_bellum
				has_law_or_variant = law_type:law_disarmed_nation
				has_law_or_variant = law_type:law_armed_neutrality
			}
		}
		localization_key = bpm_mindef_name_usa_jb
	}
	text = {
		trigger = {
			exists = c:USA
			THIS ?= c:USA
			bpm_country_is_bourgeois_republic = yes
		}
		localization_key = bpm_mindef_name_usa
	}
	text = {
		trigger = {
			OR = {
				has_law_or_variant = law_type:law_justum_bellum
				has_law_or_variant = law_type:law_disarmed_nation
				has_law_or_variant = law_type:law_armed_neutrality
			}
		}
		localization_key = bpm_mindef_name_usa_jb
	}
	text = {
		trigger = {
			always = no
		}
		fallback = yes
		localization_key = bpm_mindef_name_default
	}
}
