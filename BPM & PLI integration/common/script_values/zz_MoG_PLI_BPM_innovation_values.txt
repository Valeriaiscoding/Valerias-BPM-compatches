REPLACE:mog_pli_ino_policing = {				# Max 15
	if = {
		limit = {	# Gotta Defend your own Investments
			has_law_or_variant = law_type:law_no_police
		}
		add = 0
	}
	else_if = {
		limit = {
			OR = { 
				has_law_or_variant = law_type:law_local_police 			# Local police are ok, but prone to injustice and corruption
				has_law_or_variant = law_type:law_militarized_police	# The law is the law, it is harsh but can be wielded to advantage
			}
		}
		add = 2		# Max 10
		multiply = {
			value = institution:institution_police.investment
			min = 1
		}
	}
	else_if = {
		limit = {	# One of the biggest contributors to a healthy investment environment
			OR = {
				has_law_or_variant = law_type:law_dedicated_police
				has_law_or_variant = law_type:law_elected_police
			}
		}
		add = 3		# Max 15
		multiply = {
			value = institution:institution_police.investment
			min = 1
		}
	}
	else = {	# For Random Modded laws
		add = mog_pli_ino_policing_modded		# Max 5 by default
		owner.institution:institution_police ?= {
			multiply = {
				value = investment	# institution:institution_police.investment # already in institution scope
				min = 1
			}
		}
	}
}
