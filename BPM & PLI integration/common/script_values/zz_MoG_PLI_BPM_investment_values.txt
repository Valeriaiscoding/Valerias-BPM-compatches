REPLACE:mog_pli_investment_increment = {
	value = 0
	if = {
		limit = { mog_pli_investment_target_trend_buffer > var:mog_pli_investment_current_value } # Target below current state.
		add = {
			add = -1.5		# -ve because then multiplied by -ve from inverse %
			multiply = {
				add = var:mog_pli_investment_current_value			# (current/target)-1 = bigger absolute
				divide = mog_pli_investment_target_trend_buffer		# value when further from the target
				add = -1
			}
			desc = PLI_BASE_TREND
			format = PLI_3DECIMAL_SCRIPT_VALUE_FORMAT
		}
	}
	# Accelerators/Decelerators

	add = {
		owner.institution:institution_police ?= {
			add = 0.06
			multiply = investment
		}
		desc = PLI_POLICE_INSTITUTION		# Max = 0.3
		format = PLI_2DECIMAL_SCRIPT_VALUE_FORMAT
	}
	add = {
		owner.institution:institution_economy ?= {
			add = 0.1 
			multiply = investment
		}
		# TODO LOC
		desc = BPM_PLI_ECONOMY_INSTITUTION		# Max = 0.5
		format = PLI_2DECIMAL_SCRIPT_VALUE_FORMAT
	}

	add = {
		owner.institution:institution_schools ?= {
			add = 0.03
			multiply = investment
		}
		desc = PLI_SCHOOL_INSTITUTION		# Max = 0.15
		format = PLI_2DECIMAL_SCRIPT_VALUE_FORMAT
	}
	add = {
		owner.institution:institution_health_system ?= {
			add = 0.03
			multiply = investment
		}
		desc = PLI_HEALTH_INSTITUTION		# Max = 0.15
		format = PLI_2DECIMAL_SCRIPT_VALUE_FORMAT
	}
	add = {
		owner.institution:institution_social_security ?= {
			add = 0.08
			multiply = investment
		}
		desc = PLI_SOCSEC_INSTITUTION		# Max = 0.4
		format = PLI_2DECIMAL_SCRIPT_VALUE_FORMAT
	}

	# Decelerators
	add = {		# New Investment has higher overheads
		owner.institution:institution_workplace_safety ?= {
			add = -0.1
			multiply = investment
		}
		desc = PLI_OHS_INSTITUTION			# Max = -0.5
		format = PLI_2DECIMAL_SCRIPT_VALUE_FORMAT
	}
	add = {		# New Investment has either sovereign risk or more noise to filter
		owner.institution:institution_home_affairs ?= {
			add = -0.03
			multiply = investment
		}
		desc = PLI_HAFFAIRS_INSTITUTION		# Max = -0.15
		format = PLI_2DECIMAL_SCRIPT_VALUE_FORMAT
	}
	add = {		# New Investment has either sovereign risk or more noise to filter
		owner.institution:institution_culture ?= {
			add = -0.03
			multiply = investment
		}
		#TODO LOC
		desc = BPM_PLI_CULTURE_INSTITUTION		# Max = -0.15
		format = PLI_2DECIMAL_SCRIPT_VALUE_FORMAT
	}
	
	# Cenralization
	add = {
		owner.institution:institution_centralization ?= {
			value = -3
			add = investment
			multiply = 0.1
		}
		# TODO LOC
		desc = BPM_PLI_CENTRALIZATION_INSTITUTION		# Max = 0.2
		format = PLI_2DECIMAL_SCRIPT_VALUE_FORMAT
	}

	# Stability
	if = {
		limit = { owner = { has_variable = var:bpm_political_stability } }
		add = {
			# TODO LOC
			value = owner.var:bpm_political_stability
			subtract = 50
			multiply = -0.007 # -0.35 to 0.35
			desc = "PLI_BPM_POLITICAL_STABILITY"
		}
	}


	min = 0.1
}

mog_pli_inv_stability = {	# Max 15 optimal value 60 
	if = {
		limit = { has_variable = var:bpm_political_stability }
		value = 15
		if = {
			limit = { var:bpm_political_stability > 60 }	
			subtract = {
				value = var:bpm_political_stability
				subtract = 60
				multiply = 0.25
			}
		}
		else_if = {
			limit = { var:bpm_political_stability < 60 }
			subtract = {
				value = 60
				subtract = var:bpm_political_stability
				multiply = 0.5
			}
		}
	}
	else = {
		value = 10	
	}
}
