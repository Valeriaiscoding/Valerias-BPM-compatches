mog_pli_inno_stablity = { # Max 20 	best value 30 
	if = {
		limit = { has_variable = var:bpm_political_stability }
		value = 20

		if = {
			limit = {
				var:bpm_political_stability > 30
			}
			subtract = {
				value = var:bpm_political_stability
				subtract = 30
				multiply = 0.5
			}
		}
		else_if = {
			limit = {
				var:bpm_political_stability < 30
			}
			subtract = {
				value = 30
				subtract = var:bpm_political_stability
				multiply = 0.5
			}
		}
	}
	else = {
		value = 14
	}
	min = 0
}
