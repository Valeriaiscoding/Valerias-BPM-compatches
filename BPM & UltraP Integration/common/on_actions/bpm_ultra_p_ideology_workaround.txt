on_law_activated = {
	on_actions = {
		bpm_ultra_p_slavery
	}
}

bpm_ultra_p_slavery = {
	effect = {
		if = {
			limit = { type = law_type:law_slavery_banned }
			owner = {
				if = {
					limit = { c:BRZ ?= this }
					ig:ig_petty_bourgeoisie ?= { 
						if = {
							limit = { bpm_ig_is_pro_slavery = yes }
							bpm_remove_all_eco_ideologies = yes
							bpm_reorder_ideologies = yes
						}
					}
					ig:ig_intelligentsia ?= { 
						if = {
							limit = { 
								bpm_ig_is_pro_slavery = yes
							}
							bpm_remove_all_eco_ideologies = yes
							add_ideology = ideology_eco_artisan
							bpm_reorder_ideologies = yes
						}
					}
				}
			}
		}
	}
}
