bpm_ultra_p_setup_country_eco_ideologies = {
	ig:ig_intelligentsia = {
		if = {
			limit = { has_ideology = ideology:ideology_eco_placeholder }
			bpm_remove_all_eco_ideologies = yes
			add_ideology = ideology_eco_artisan
		}
	}
	ig:ig_conservatives = {
		if = {
			limit = { 
				owner = {
					bpm_country_is_latin_american = yes
				} 
				has_ideology = ideology:ideology_eco_conservative_agrarian
			}
			bpm_remove_all_eco_ideologies = yes
			add_ideology = ideology_eco_export_agrarian
		}
	}
	
}

bpm_ultra_p_setup_country_dop_ideologies = {
	ig:ig_rural_folk = {
		if = {
			limit = { has_ideology = ideology:ideology_dop_placeholder }
			bpm_remove_all_dop_ideologies = yes
			add_ideology = ideology_dop_particularist
		}
	}
}

REPLACE:bpm_setup_country_wel_ideologies = {
	if = {
		limit = { bpm_country_is_in_new_world = yes }
		if = {
			limit = { country_has_state_religion = rel:catholic }
			ig:ig_reactionaries ?= { 
				if = {
					limit = { has_ideology = ideology:ideology_wel_no_welfare }
				}
				remove_ideology = ideology_wel_no_welfare
				add_ideology = ideology_wel_religious_welfare
			}
		}
	}
	bpm_ultra_p_setup_country_eco_ideologies = yes
	bpm_ultra_p_setup_country_wel_ideologies = yes
}
